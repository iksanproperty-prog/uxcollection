Pengaturan Toko & Payment Gateway

Struktur Database MySQL (`ux_collection_db.sql`)

-- UX COLLECTION MySQL Database Schema for PHP 8+ cPanel Hosting
CREATE DATABASE IF NOT EXISTS `ux_collection_db`;
USE `ux_collection_db`;

CREATE TABLE `categories` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `name` VARCHAR(100) NOT NULL,
  `slug` VARCHAR(100) NOT NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE `products` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `category_id` INT,
  `title` VARCHAR(255) NOT NULL,
  `price` DECIMAL(12,2) NOT NULL,
  `promo_price` DECIMAL(12,2) DEFAULT NULL,
  `stock` INT DEFAULT 10,
  `badge` VARCHAR(50),
  `image` VARCHAR(255),
  `description` TEXT,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (`category_id`) REFERENCES `categories`(`id`) ON DELETE SET NULL
);

CREATE TABLE `orders` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `order_number` VARCHAR(50) UNIQUE NOT NULL,
  `customer_name` VARCHAR(100) NOT NULL,
  `whatsapp` VARCHAR(30) NOT NULL,
  `address` TEXT NOT NULL,
  `province` VARCHAR(100),
  `city` VARCHAR(100),
  `postal_code` VARCHAR(20),
  `subtotal` DECIMAL(12,2),
  `shipping_cost` DECIMAL(12,2),
  `total_amount` DECIMAL(12,2),
  `payment_method` VARCHAR(50),
  `status` ENUM('Menunggu Pembayaran','Lunas','Diproses','Sedang Digambar','Sedang Dicetak','Packing','Dikirim','Selesai') DEFAULT 'Menunggu Pembayaran',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE `payments` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `order_id` INT NOT NULL,
  `qr_payload` TEXT,
  `transaction_id` VARCHAR(100),
  `payment_status` VARCHAR(50) DEFAULT 'PENDING',
  `paid_at` DATETIME NULL,
  FOREIGN KEY (`order_id`) REFERENCES `orders`(`id`) ON DELETE CASCADE
);

CREATE TABLE `settings` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `setting_key` VARCHAR(100) UNIQUE NOT NULL,
  `setting_value` TEXT
);
